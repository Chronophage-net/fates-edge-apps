API_BASE = os.getenv("FATES_EDGE_API_BASE", "https://your-fates-edge-server.com/api")
API_KEY = os.getenv("FATES_EDGE_API_KEY", "YOUR_API_KEY")
DEFAULT_ROOM = os.getenv("FATES_EDGE_ROOM", "YOUR_ROOM_CODE")

# ============================================================

# ============================================================

def make_request(endpoint, method="GET", data=None):
    try:
        url = f"{API_BASE}{endpoint}"
        headers = {"x-api-key": API_KEY, "Content-Type": "application/json"}
        if method == "GET":
            response = requests.get(url, headers=headers)
        else:
            response = requests.post(url, json=data, headers=headers)
        response.raise_for_status()
        return response.json()
    except Exception as e:
        return {"error": str(e)}

def get_room_code():
    return DEFAULT_ROOM

def get_client_id_by_name(name):
    room_code = get_room_code()
    clients = make_request(f"/rooms/{room_code}/clients").get("clients", [])
    for c in clients:
        if c.get("name", "").lower() == name.lower():
            return c["id"]
    return None

def format_character(c):
    name = c.get("name", "Unknown")
    harm = c.get("harm", 0)
    fatigue = c.get("fatigue", 0)
    boons = c.get("boons", 0)
    tier = c.get("tier", 1)
    heritage = c.get("heritage", "")
    background = c.get("background", "")
    patron = c.get("patron", "")
    attrs = c.get("attributes", {})
    skills = c.get("skills", {})
    output = f"**{name}**"
    if heritage: output += f" ({heritage})"
    output += f"\n  ❤️ {harm}  ⚡ {fatigue}  🪙 {boons}  Tier {tier}"
    if attrs:
        attr_str = ", ".join(f"{k}:{v}" for k,v in attrs.items())
        output += f"\n  **Attributes:** {attr_str}"
    if skills:
        skill_str = ", ".join(f"{k}:{v}" for k,v in skills.items())
        output += f"\n  **Skills:** {skill_str}"
    if background: output += f"\n  **Background:** {background}"
    if patron: output += f"\n  **Patron:** {patron}"
    return output

def parse_ref(ref_str):
    # Try to parse as int, else keep as string
    try:
        return int(ref_str)
    except ValueError:
        return ref_str

if cmd == "draw":
    count = min(int(args[1]) if len(args) > 1 else 1, 5)
    region = args[2] if len(args) > 2 else "Acasia"
    result = make_request(f"/rooms/{room}/deck/draw", "POST", {"count": count, "region": region})
    if result.get("error"):
        return f"❌ {result['error']}"
    cards = result.get("cards", [])
    synthesis = result.get("synthesis", "")
    card_list = "\n".join([f"• {c['rankName']} of {c['suitName']}" for c in cards])
    return f"🃏 **Drew {count} from {region}**\n{card_list}\n\n{synthesis}"

elif cmd == "crown":
    region = args[1] if len(args) > 1 else "Acasia"
    result = make_request(f"/rooms/{room}/deck/crown", "POST", {"region": region})
    if result.get("error"):
        return f"❌ {result['error']}"
    positions = result.get("result", {}).get("positions", [])
    output = "👑 **Crown Spread**\n"
    for pos in positions:
        output += f"{pos['icon']} **{pos['label']}**: {pos['meaning']}\n"
    return output

elif cmd == "shuffle":
    result = make_request(f"/rooms/{room}/deck/shuffle", "POST")
    if result.get("error"):
        return f"❌ {result['error']}"
    return "🔀 Deck shuffled."

# ============================================================
# CHARACTER COMMANDS
# ============================================================

elif cmd == "chars":
    if len(args) < 2:
        return "❌ Usage: `!fe chars list|view <name>|update <name> <field> <value>`"
    sub = args[1].lower()
    if sub == "list":
        result = make_request(f"/rooms/{room}/characters")
        if result.get("error"):
            return f"❌ {result['error']}"
        chars = result.get("characters", [])
        if not chars:
            return "📭 No characters in the room."
        output = "👥 **Characters**\n"
        for c in chars:
            name = c.get("name", "Unknown")
            harm = c.get("harm", 0)
            fatigue = c.get("fatigue", 0)
            boons = c.get("boons", 0)
            output += f"  {name} (❤️{harm} ⚡{fatigue} 🪙{boons})\n"
        return output
    elif sub == "view":
        if len(args) < 3:
            return "❌ Usage: `!fe chars view <name>`"
        name = " ".join(args[2:])
        result = make_request(f"/rooms/{room}/characters/{name}")
        if result.get("error"):
            return f"❌ {result['error']}"
        return format_character(result)
    elif sub == "update":
        if len(args) < 5:
            return "❌ Usage: `!fe chars update <name> <field> <value>`"
        name = args[2]
        field = args[3]
        value = " ".join(args[4:])  # allow spaces
        try:
            value = int(value)
        except:
            try:
                value = float(value)
            except:
                pass
        updates = {field: value}
        payload = {"updates": {name: updates}}
        result = make_request(f"/rooms/{room}/characters/update", "POST", payload)
        if result.get("error"):
            return f"❌ {result['error']}"
        return f"✅ **{name}** `{field}` updated to `{value}`."

elif cmd == "grid":
    wb = make_request(f"/rooms/{room}/whiteboard")
    if wb.get("error"):
        return f"❌ {wb['error']}"
    gc = wb.get("gridCombat", {})
    enabled = gc.get("enabled", False)
    grid_type = gc.get("gridType", "square")
    cell_size = gc.get("cellSize", 40)
    tokens = gc.get("tokens", [])
    output = "⚔️ **Grid Combat Status**\n"
    output += f"  Enabled: {'✅' if enabled else '❌'}\n"
    if enabled:
        output += f"  Grid Type: {grid_type}\n"
        output += f"  Cell Size: {cell_size}\n"
        output += f"  Tokens: {len(tokens)}\n"
        if tokens:
            output += "    " + ", ".join([f"{t.get('name', '?')} ({t.get('x',0)},{t.get('y',0)})" for t in tokens[:5]]) + ("..." if len(tokens)>5 else "")
    return output

elif cmd == "wb":
    wb = make_request(f"/rooms/{room}/whiteboard")
    if wb.get("error"):
        return f"❌ {wb['error']}"
    drawings = len(wb.get("drawings", []))
    notes = len(wb.get("notes", []))
    images = len(wb.get("images", []))
    output = "📋 **Whiteboard Summary**\n"
    output += f"  Drawings: {drawings}\n"
    output += f"  Notes: {notes}\n"
    output += f"  Images: {images}\n"
    gc = wb.get("gridCombat", {})
    if gc.get("enabled"):
        output += f"  Grid Combat: Active ({gc.get('gridType','square')}, {len(gc.get('tokens',[]))} tokens)"
    return output

elif cmd == "gm":
    if len(args) < 2:
        return "❌ Usage: `!fe gm status|request|approve <player>`"
    sub = args[1].lower()
    if sub == "status":
        clients = make_request(f"/rooms/{room}/clients").get("clients", [])
        gm = next((c for c in clients if c.get("role") == "gm"), None)
        if gm:
            return f"👑 **GM**: {gm['name']}\n👥 **Players**: {len(clients)}"
        else:
            return "👑 No GM currently."
    elif sub == "request":
        return "❌ GM request via REST API is not implemented. Use WebSocket bot."
    elif sub == "approve":
        return "❌ GM approve via REST API is not implemented."
    else:
        return "❌ Unknown GM subcommand."

elif cmd == "players":
    clients = make_request(f"/rooms/{room}/clients").get("clients", [])
    if not clients:
        return "📭 No players in the room."
    output = "👥 **Players in Room**\n"
    for c in clients:
        output += f"`{c['id']}` - **{c['name']}** ({c['role']})\n"
    return output

elif cmd == "kick":
    if len(args) < 2:
        return "❌ Usage: `!fe kick <player name or ID> [reason]`"
    target = args[1]
    reason = " ".join(args[2:]) if len(args) > 2 else "Kicked by Avrae admin"
    client_id = target
    if not (target.startswith("ws-") or len(target) == 20):
        client_id = get_client_id_by_name(target)
        if not client_id:
            return f"❌ Player '{target}' not found."
    result = make_request(f"/rooms/{room}/clients/{client_id}/kick", "POST", {"reason": reason})
    if result.get("error"):
        return f"❌ {result['error']}"
    return f"👢 Kicked {client_id} ({target})."

elif cmd == "ban":

    client_id = target
    if not (target.startswith("ws-") or len(target) == 20):
        client_id = get_client_id_by_name(target)
        if not client_id:
            return f"❌ Player '{target}' not found."

    if result.get("error"):

    if len(args) < 2:

elif cmd == "adventure":
    if len(args) < 2:
        return "❌ Usage: `!fe adventure load|scene|encounter|timer|log|status|reference|reset`"
    sub = args[1].lower()
    sub_args = args[2:]

    # --- LOAD ---
    if sub == "load":
        if not sub_args:
            return "❌ Usage: `!fe adventure load <moduleId>`"
        module_id = sub_args[0]
        result = make_request(f"/rooms/{room}/adventure/load", "POST", {"moduleId": module_id})
        if result.get("error"):
            return f"❌ {result['error']}"
        return f"📖 Loaded adventure **{module_id}**."

    # --- SCENE ---
    elif sub == "scene":
        target = {}
        if sub_args:
            # try to parse actIndex, sceneIndex
            if len(sub_args) >= 1 and sub_args[0].isdigit():
                target["actIndex"] = int(sub_args[0])
                if len(sub_args) >= 2 and sub_args[1].isdigit():
                    target["sceneIndex"] = int(sub_args[1])
        result = make_request(f"/rooms/{room}/adventure/scene", "POST", target)
        if result.get("error"):
            return f"❌ {result['error']}"
        state = result
        act_title = state.get("currentAct", {}).get("title", "Unknown")
        scene_title = state.get("currentScene", {}).get("title", "Unknown")
        return f"🎭 Advanced to scene **{scene_title}** in act **{act_title}**."

    # --- ENCOUNTER ---
    elif sub == "encounter":
        if len(sub_args) < 2:
            return "❌ Usage: `!fe adventure encounter start <ref>` or `resolve <outcome> [notes]`"
        enc_sub = sub_args[0].lower()
        enc_args = sub_args[1:]
        if enc_sub == "start":
            if not enc_args:
                return "❌ Usage: `!fe adventure encounter start <ref>` (ref = index or name/creatureId)"
            ref = parse_ref(enc_args[0])
            result = make_request(f"/rooms/{room}/adventure/encounter/start", "POST", {"ref": ref})
            if result.get("error"):
                return f"❌ {result['error']}"
            return f"⚔️ Encounter started: **{ref}**"
        elif enc_sub == "resolve":
            if not enc_args:
                return "❌ Usage: `!fe adventure encounter resolve <outcome> [notes]` (outcome: clean|partial|miss)"
            outcome = enc_args[0].lower()
            if outcome not in ["clean", "partial", "miss"]:
                return "❌ Outcome must be `clean`, `partial`, or `miss`"
            notes = " ".join(enc_args[1:]) if len(enc_args) > 1 else ""
            result = make_request(f"/rooms/{room}/adventure/encounter/resolve", "POST", {"outcome": outcome, "notes": notes})
            if result.get("error"):
                return f"❌ {result['error']}"
            return f"⚔️ Encounter resolved as **{outcome}**."
        else:
            return "❌ Unknown encounter subcommand. Use `start` or `resolve`."

    # --- TIMER ---
    elif sub == "timer":
        if len(sub_args) < 1:
            return "❌ Usage: `!fe adventure timer <name> [amount] [scope]`"
        name = sub_args[0]
        amount = int(sub_args[1]) if len(sub_args) > 1 and sub_args[1].lstrip('-').isdigit() else 1
        scope = sub_args[2] if len(sub_args) > 2 else "scene"
        if scope not in ["scene", "campaign"]:
            scope = "scene"
        result = make_request(f"/rooms/{room}/adventure/timer", "POST", {"ref": name, "amount": amount, "scope": scope})
        if result.get("error"):
            return f"❌ {result['error']}"
        return f"⏱️ Timer **{name}** ticked by {amount} ({scope} scope)."

    # --- LOG ---
    elif sub == "log":
        if not sub_args:
            return "❌ Usage: `!fe adventure log <text> [author]`"
        text = sub_args[0]
        author = sub_args[1] if len(sub_args) > 1 else "Avrae"
        result = make_request(f"/rooms/{room}/adventure/log", "POST", {"text": text, "author": author})
        if result.get("error"):
            return f"❌ {result['error']}"
        return f"📝 Logged beat: \"{text}\""

    # --- STATUS ---
    elif sub == "status":
        result = make_request(f"/rooms/{room}/adventure")
        if result.get("error"):
            return f"❌ {result['error']}"
        if not result.get("moduleId"):
            return "📭 No adventure loaded in this room."
        state = result
        title = state.get("title", state.get("moduleId", "Unknown"))
        status = state.get("status", "unknown")
        act_title = state.get("currentAct", {}).get("title", "None")
        scene_title = state.get("currentScene", {}).get("title", "None")
        output = f"📖 **{title}**\n  Status: {status}\n  Act: {act_title}\n  Scene: {scene_title}"
        if state.get("activeEncounter"):
            enc = state["activeEncounter"]
            enc_name = enc.get("name") or enc.get("creatureId") or "Encounter"
            dv = enc.get("dv", "?")
            pos = enc.get("position", "Controlled")
            output += f"\n  ⚔️ Encounter: {enc_name} (DV {dv}, {pos})"
        if state.get("campaignTimers"):
            timers = ", ".join([f"{t['name']}: {t.get('current',0)}/{t['segments']}" for t in state["campaignTimers"]])
            output += f"\n  ⏱️ Campaign Timers: {timers}"
        if state.get("log") and len(state["log"]) > 0:
            last = state["log"][-1]
            output += f"\n  📜 Last log: {last.get('message') or last.get('type')}"
        return output

    # --- REFERENCE ---
    elif sub == "reference":
        result = make_request(f"/rooms/{room}/adventure/reference")
        if result.get("error"):
            return f"❌ {result['error']}"
        if not result.get("moduleId"):
            return "📭 No adventure loaded."
        output = f"📚 **Reference for {result['moduleId']}**\n"
        if result.get("bestiary"):
            best = result["bestiary"][:5]
            output += f"🐉 Bestiary ({len(result['bestiary'])}): " + ", ".join([f"{b['name']} (TL{b.get('tl','?')})" for b in best]) + ("..." if len(result['bestiary'])>5 else "") + "\n"
        if result.get("npcs"):
            npcs = result["npcs"][:5]
            output += f"👤 NPCs ({len(result['npcs'])}): " + ", ".join([f"{n['name']} ({n.get('role','NPC')})" for n in npcs]) + ("..." if len(result['npcs'])>5 else "") + "\n"
        if result.get("locations"):
            locs = result["locations"][:5]
            output += f"📍 Locations ({len(result['locations'])}): " + ", ".join([l['name'] for l in locs]) + ("..." if len(result['locations'])>5 else "") + "\n"
        if result.get("factions"):
            facs = result["factions"]
            output += f"⚑ Factions ({len(facs)}): " + ", ".join([f['name'] for f in facs]) + "\n"
        if result.get("notes"):
            output += f"📝 Notes: {result['notes'][:200]}{'...' if len(result['notes'])>200 else ''}"
        return output

    # --- RESET ---
    elif sub == "reset":
        result = make_request(f"/rooms/{room}/adventure/reset", "POST")
        if result.get("error"):
            return f"❌ {result['error']}"
        return "🔄 Adventure reset to initial state."

    else:
        return "❌ Unknown adventure subcommand. Use: load, scene, encounter, timer, log, status, reference, reset."
