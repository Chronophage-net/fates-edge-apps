!alias fates-edge <drac2>
args = &ARGS&
if not args or args[0] == "help":
    return 'echo Fate\'s Edge local dice: `!fates-edge roll POOL` (1–30 d10). Apply outcome rules at the table. Avrae aliases cannot connect to the Fate\'s Edge server; use the dedicated Discord bot for live room commands.'
if args[0] != "roll" or len(args) != 2 or not args[1].isdigit():
    return 'echo Usage: !fates-edge roll POOL (1–30)'
pool = int(args[1])
if pool < 1 or pool > 30:
    return 'echo Pool must be between 1 and 30 dice.'
result = vroll(f'{pool}d10')
return f'echo Fate\'s Edge dice — {result}. Apply successes and Story Beats using your table rules.'
</drac2>
